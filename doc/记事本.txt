EMA 的作用
emaShort 和 emaLong 用于趋势判断,是技术分析中最常用的指标之一：

指标	典型周期	作用
emaShort	10-20 周期	反映短期价格趋势,更敏感
emaLong	30-50 周期	反映长期价格趋势,更稳定
交叉信号：

emaShort 上穿 emaLong → 金叉 → 上涨趋势信号

emaShort 下穿 emaLong → 死叉 → 下跌趋势信号

EMA vs SMA 对比
特性	EMA	SMA
计算方式	指数加权,近期权重更高	简单平均,所有周期等权
响应速度	快,对价格变化敏感	慢,有滞后
适用场景	趋势跟踪、短线交易	判断大方向、过滤噪音
平滑度	较平滑	更平滑

价格:    100 → 101 → 102 → 103 → 104 → 105 → 106 → 107 → 108 → 109

SMA(5):  100 → 101 → 102 → 103 → 104 → 105 → 106 → 107 → 108 → 109
EMA(5):  100 → 100.6→101.3→102.1→102.9→103.8→104.7→105.6→106.6→107.6

当价格从100涨到109：
- EMA 从 100 → 107.6（上涨7.6）
- SMA 从 100 → 109（上涨9）

说明：EMA 更平滑,但 SMA 更贴近实际价格


SignalPeriod 的作用
信号线（Signal Line） 是 EMA 的二次平滑,用于生成更可靠的交易信号：

概念	说明	典型周期
快线 (Fast EMA)	短期EMA,反应灵敏	10-20
慢线 (Slow EMA)	长期EMA,趋势确认	30-50
信号线 (Signal Line)	快线的EMA,用于过滤假信号	5-9
典型应用（MACD指标）：

MACD线 = 快线 - 慢线

信号线 = MACD线的EMA

金叉：MACD线上穿信号线 → 买入信号

死叉：MACD线下穿信号线 → 卖出信号

// calculateWindowTrend 计算时间窗口内的趋势（秒级数据优化版）
func (s *GridStrategy) calculateWindowTrend(prices []float64, volumes []int64, timestamps []int64) string {
	windowSize := len(prices)
	if windowSize < s.appConfig.VolumeSmoothPeriod {
		return "SIDEWAYS"
	}

	// 1. 价格趋势分析（使用线性回归）
	priceTrend := s.calculatePriceTrend(prices)
	
	// 2. 成交量趋势分析（使用加权移动平均）
	volumeTrend := s.calculateVolumeTrend(volumes)
	
	// 3. 量价配合度分析
	correlation := s.calculatePriceVolumeCorrelation(prices, volumes)
	
	// 4. 趋势强度计算
	trendStrength := math.Abs(priceTrend) * (0.4 + 0.6*correlation)
	
	// 5. 动态阈值（基于波动率）
	volatility := s.calculateVolatility(prices)
	threshold := s.getDynamicThreshold(volatility)
	
	// 详细日志
	logger.Debug("[%s] 趋势分析: 价格趋势=%.4f%%, 成交量趋势=%.4f%%, 量价相关度=%.2f, 趋势强度=%.4f, 波动率=%.2f%%, 阈值=%.4f",
		s.name, priceTrend*100, volumeTrend, correlation, trendStrength, volatility*100, threshold)
	
	// 判断趋势
	if priceTrend > threshold && correlation > 0.3 {
		return "UP"
	} else if priceTrend < -threshold && correlation > 0.3 {
		return "DOWN"
	} else if priceTrend > threshold/2 && volumeTrend > 0.1 {
		// 弱上涨但成交量配合
		return "UP_WEAK"
	} else if priceTrend < -threshold/2 && volumeTrend > 0.1 {
		// 弱下跌但成交量配合
		return "DOWN_WEAK"
	}
	
	return "SIDEWAYS"
}

// calculatePriceTrend 计算价格趋势（使用线性回归斜率）
func (s *GridStrategy) calculatePriceTrend(prices []float64) float64 {
	n := float64(len(prices))
	if n < 2 {
		return 0
	}
	
	// 线性回归计算斜率
	var sumX, sumY, sumXY, sumX2 float64
	for i, price := range prices {
		x := float64(i)
		y := price
		sumX += x
		sumY += y
		sumXY += x * y
		sumX2 += x * x
	}
	
	// 斜率 = (n*Σxy - Σx*Σy) / (n*Σx² - (Σx)²)
	denominator := n*sumX2 - sumX*sumX
	if denominator == 0 {
		return 0
	}
	slope := (n*sumXY - sumX*sumY) / denominator
	
	// 归一化斜率（相对于平均价格）
	avgPrice := sumY / n
	if avgPrice == 0 {
		return 0
	}
	normalizedSlope := slope / avgPrice
	
	return normalizedSlope
}

// calculateVolumeTrend 计算成交量趋势（加权移动平均）
func (s *GridStrategy) calculateVolumeTrend(volumes []int64) float64 {
	if len(volumes) < 10 {
		return 0
	}
	
	// 计算近期加权平均
	recentWeighted := 0.0
	recentWeight := 0.0
	// 计算远期加权平均
	oldWeighted := 0.0
	oldWeight := 0.0
	
	// 近期权重更高（最近40%的数据权重60%）
	recentStart := len(volumes) * 6 / 10
	for i := recentStart; i < len(volumes); i++ {
		weight := float64(i - recentStart + 1)
		recentWeighted += float64(volumes[i]) * weight
		recentWeight += weight
	}
	
	// 远期数据（前40%）
	oldEnd := len(volumes) * 4 / 10
	for i := 0; i < oldEnd; i++ {
		weight := float64(oldEnd - i)
		oldWeighted += float64(volumes[i]) * weight
		oldWeight += weight
	}
	
	if recentWeight == 0 || oldWeight == 0 {
		return 0
	}
	
	avgRecent := recentWeighted / recentWeight
	avgOld := oldWeighted / oldWeight
	
	if avgOld == 0 {
		return 0
	}
	
	// 成交量变化率（正数表示放量，负数表示缩量）
	volumeChange := (avgRecent - avgOld) / avgOld
	
	// 限制变化范围在[-1, 1]之间
	if volumeChange > 1 {
		volumeChange = 1
	} else if volumeChange < -1 {
		volumeChange = -1
	}
	
	return volumeChange
}

// calculatePriceVolumeCorrelation 计算量价相关系数
func (s *GridStrategy) calculatePriceVolumeCorrelation(prices []float64, volumes []int64) float64 {
	n := len(prices)
	if n < 5 {
		return 0
	}
	
	// 计算价格变化率和成交量变化率序列
	priceChanges := make([]float64, n-1)
	volumeChanges := make([]float64, n-1)
	
	for i := 1; i < n; i++ {
		if prices[i-1] != 0 {
			priceChanges[i-1] = (prices[i] - prices[i-1]) / prices[i-1]
		}
		if volumes[i-1] > 0 {
			volumeChanges[i-1] = float64(volumes[i]-volumes[i-1]) / float64(volumes[i-1])
		} else if volumes[i] > 0 {
			volumeChanges[i-1] = 1
		}
		
		// 限制极端值
		if priceChanges[i-1] > 0.1 {
			priceChanges[i-1] = 0.1
		} else if priceChanges[i-1] < -0.1 {
			priceChanges[i-1] = -0.1
		}
		if volumeChanges[i-1] > 2 {
			volumeChanges[i-1] = 2
		} else if volumeChanges[i-1] < -1 {
			volumeChanges[i-1] = -1
		}
	}
	
	// 计算相关系数
	return s.calculateCorrelation(priceChanges, volumeChanges)
}

// calculateCorrelation 计算两个序列的相关系数
func (s *GridStrategy) calculateCorrelation(x, y []float64) float64 {
	n := float64(len(x))
	if n < 2 {
		return 0
	}
	
	var sumX, sumY, sumXY, sumX2, sumY2 float64
	for i := 0; i < len(x); i++ {
		sumX += x[i]
		sumY += y[i]
		sumXY += x[i] * y[i]
		sumX2 += x[i] * x[i]
		sumY2 += y[i] * y[i]
	}
	
	denominator := math.Sqrt((n*sumX2 - sumX*sumX) * (n*sumY2 - sumY*sumY))
	if denominator == 0 {
		return 0
	}
	
	correlation := (n*sumXY - sumX*sumY) / denominator
	
	// 限制范围在[-1, 1]
	if correlation > 1 {
		correlation = 1
	} else if correlation < -1 {
		correlation = -1
	}
	
	return correlation
}

// calculateVolatility 计算波动率
func (s *GridStrategy) calculateVolatility(prices []float64) float64 {
	if len(prices) < 2 {
		return 0
	}
	
	// 计算收益率序列
	returns := make([]float64, len(prices)-1)
	for i := 1; i < len(prices); i++ {
		if prices[i-1] != 0 {
			returns[i-1] = (prices[i] - prices[i-1]) / prices[i-1]
		}
	}
	
	// 计算标准差
	mean := 0.0
	for _, r := range returns {
		mean += r
	}
	mean /= float64(len(returns))
	
	variance := 0.0
	for _, r := range returns {
		diff := r - mean
		variance += diff * diff
	}
	variance /= float64(len(returns))
	
	return math.Sqrt(variance)
}

// getDynamicThreshold 获取动态阈值
func (s *GridStrategy) getDynamicThreshold(volatility float64) float64 {
	// 基础阈值0.1%（日内的价格波动）
	baseThreshold := 0.001
	
	// 根据波动率调整阈值
	if volatility > 0.003 {
		// 高波动市场，提高阈值
		return baseThreshold * 2
	} else if volatility < 0.001 {
		// 低波动市场，降低阈值
		return baseThreshold * 0.5
	}
	
	return baseThreshold
}

func (s *GridStrategy) calculateWindowTrend(prices []float64, volumes []int64, timestamps []int64) string {
	windowSize := len(prices)
	if windowSize < s.appConfig.VolumeSmoothPeriod {
		return "SIDEWAYS"
	}
	
	// 1. 价格变化率（首尾比较）

	// 2. 成交量变化率（加权）
	volumeChange := s.calculateWeightedVolumeChange(volumes)
	
	// 3. 价格方向一致性（多数原则）
	direction := s.calculatePriceDirection(prices)
	
	// 4. 综合评分
	score := 0.0
	
	// 价格因素（权重50%）
	if priceChange > 0.001 {
		score += 0.5
	} else if priceChange < -0.001 {
		score -= 0.5
	}
	
	// 成交量因素（权重30%）
	if volumeChange > 0.1 {
		score += 0.3
	} else if volumeChange < -0.1 {
		score -= 0.3
	}
	
	// 方向一致性（权重20%）
	if direction > 0.6 {
		score += 0.2
	} else if direction < -0.6 {
		score -= 0.2
	}
	
	// 判断趋势
	if score > 0.4 {
		return "UP"
	} else if score < -0.4 {
		return "DOWN"
	} else if score > 0.2 {
		return "UP_WEAK"
	} else if score < -0.2 {
		return "DOWN_WEAK"
	}
	
	return "SIDEWAYS"
}

// calculateWeightedVolumeChange 计算加权成交量变化
func (s *GridStrategy) calculateWeightedVolumeChange(volumes []int64) float64 {
	n := len(volumes)
	if n < 20 {
		return 0
	}
	
	// 近期数据（最近30%）
	recentStart := n * 7 / 10
	var recentSum int64
	for i := recentStart; i < n; i++ {
		recentSum += volumes[i]
	}
	avgRecent := float64(recentSum) / float64(n-recentStart)
	
	// 远期数据（前30%）
	oldEnd := n * 3 / 10
	var oldSum int64
	for i := 0; i < oldEnd; i++ {
		oldSum += volumes[i]
	}
	avgOld := float64(oldSum) / float64(oldEnd)
	
	if avgOld == 0 {
		return 0
	}
	
	return (avgRecent - avgOld) / avgOld
}

// calculatePriceDirection 计算价格方向一致性
func (s *GridStrategy) calculatePriceDirection(prices []float64) float64 {
	if len(prices) < 2 {
		return 0
	}
	
	upCount := 0
	downCount := 0
	
	for i := 1; i < len(prices); i++ {
		if prices[i] > prices[i-1] {
			upCount++
		} else if prices[i] < prices[i-1] {
			downCount++
		}
	}
	
	total := upCount + downCount
	if total == 0 {
		return 0
	}
	
	return float64(upCount-downCount) / float64(total)
}
